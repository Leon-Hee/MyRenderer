cmake_minimum_required(VERSION 3.16)

project(MyRenderer)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

add_executable(
    MyRenderer
    src/main.cpp
    src/Math/Vector.cpp
    src/Transform/Transform.cpp
    src/DepthBuffer/Depthbuffer.cpp
    src/Framebuffer/Framebuffer.cpp
    src/Rasterizer/Rasterizer.cpp
    src/Triangles/Triangles.cpp
    src/Transform/Viewport.cpp
    src/Render/Render.cpp
    src/Transform/Clip.cpp
    src/Shader/Fragment.cpp
    src/Shader/Light.cpp
    src/Shader/Shading.cpp
)

target_include_directories(
    MyRenderer
    PRIVATE
        ${PROJECT_SOURCE_DIR}/include
)
