cmake_minimum_required(VERSION 3.16)

project(MyRenderer)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

add_executable(
    MyRenderer
    src/main.cpp
    src/DepthBuffer/Depthbuffer.cpp
    src/Framebuffer/Framebuffer.cpp
    src/Math/Vector.cpp
    src/Rasterizer/RasterizerTriangles.cpp
    src/Triangles/Triangles.cpp
)

target_include_directories(
    MyRenderer
    PRIVATE
        ${PROJECT_SOURCE_DIR}/include
)
